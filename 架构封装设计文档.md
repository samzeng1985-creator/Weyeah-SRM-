## 3. 数据封装设计

### 3.1 数据对象分层

```
DO (Data Object)       - 数据层对象，与数据库表1:1对应
├── 用于数据存储
├── 与数据库映射
├── 字段与表一一对应
└── 不可直接返回给前端

DTO (Data Transfer Object)  - 数据传输对象
├── 用于服务间传输
├── 业务含义明确
└── 不一定是完整的DO

VO (View Object)      - 视图展示对象
├── 用于返回给前端
├── 定制化字段
└── 可能来自多个DO组合

ReqVO (Request VO)    - 请求对象
├── 接收前端参数
├── 参数校验
└── 不包含敏感字段
```

```java
/**
 * 分页结果基类
 */
@Data
public class PageResult<T> implements Serializable {
    
    @Schema(description = "数据列表")
    private List<T> records;
    
    @Schema(description = "总条数")
    private long total;
    
    @Schema(description = "当前页码")
    private long pageNo;
    
    @Schema(description = "每页条数")
    private long pageSize;
    
    @Schema(description = "总页数")
    private long totalPages;
    
    public static <T> PageResult<T> of(List<T> records, long total, int pageNo, int pageSize) {
        PageResult<T> pageResult = new PageResult<>();
        pageResult.setRecords(records);
        pageResult.setTotal(total);
        pageResult.setPageNo(pageNo);
        pageResult.setPageSize(pageSize);
        pageResult.setTotalPages((total + pageSize - 1) / pageSize);
        return pageResult;
    }
    
    public static <T> PageResult<T> of(IPage<T> page) {
        return of(page.getRecords(), page.getTotal(), (int)page.getCurrent(), (int)page.getSize());
    }
}
```

```java
@Data
@NoArgsConstructor
@AllArgsConstructor
public class Result<T> implements Serializable {
    
    private static final long serialVersionUID = 1L;
    
    public static final int SUCCESS_CODE = 200;
    public static final String SUCCESS_MSG = "操作成功";
    
    @Schema(description = "状态码")
    private int code;
    
    @Schema(description = "提示信息")
    private String message;
    
    @Schema(description = "业务数据")
    private T data;
    
    public static <T> Result<T> success() {
        return success(null);
    }
    
    public static <T> Result<T> success(T data) {
        return success(SUCCESS_CODE, SUCCESS_MSG, data);
    }
    
    public static <T> Result<T> success(int code, String message, T data) {
        return new Result<>(code, message, data);
    }
    
    public static <T> Result<T> error(String message) {
        return error(500, message);
    }
    
    public static <T> Result<T> error(int code, String message) {
        return new Result<>(code, message, null);
    }
    
    public boolean isSuccess() {
        return this.code == SUCCESS_CODE;
    }
}

/**
```
```java
 * 远程服务调用的统一客户端封装
 */
public abstract class AbstractRemoteClient {
    
    protected final Logger log = LoggerFactory.getLogger(getClass());
    
    @Autowired
    private ObjectMapper objectMapper;
    
    /**
     * 处理远程调用结果，包括异常处理
     */
    protected <T> T handleResult(Result<T> result, String serviceName) {
        if (result == null) {
            log.error("调用服务 {} 返回结果为空", serviceName);
            throw new ServiceException("服务调用失败，结果为空");
        }
        
        if (!result.isSuccess()) {
            log.error("调用服务 {} 返回失败: code={}, msg={}", 
                serviceName, result.getCode(), result.getMessage());
            throw new ServiceException(result.getCode(), result.getMessage());
        }
        
        return result.getData();
    }
    
    /**
     * 记录调用日志
     */
    protected void logRequest(String serviceName, String method, Object request) {
        log.info("调用远程服务: service={}, method={}, request={}", 
            serviceName, method, toJsonString(request));
    }
    
    private String toJsonString(Object obj) {
        try {
            return objectMapper.writeValueAsString(obj);
        } catch (Exception e) {
            return obj.toString();
        }
    }
}

/**
 * Supplier 远程服务客户端
 */
@FeignClient(name = "srm-supplier", contextId = "supplierRemote")
public interface SupplierRemoteClient {
    
    @GetMapping("/system/suppliers/{id}")
    Result<SupplierVO> getSupplier(@PathVariable("id") Long id);
    
    @PostMapping("/system/suppliers")
    Result<Long> createSupplier(@RequestBody SupplierReqVO reqVO);
}

/**
 * 远程服务调用的封装
 */
@Service
public class SupplierRemoteService extends AbstractRemoteClient {
    
    @Autowired
    private SupplierRemoteClient supplierRemoteClient;
    
    public SupplierVO getSupplier(Long id) {
        logRequest("srm-supplier", "getSupplier", id);
        Result<SupplierVO> result = supplierRemoteClient.getSupplier(id);
        return handleResult(result, "srm-supplier");
    }
    
    public Long createSupplier(SupplierReqVO reqVO) {
        logRequest("srm-supplier", "createSupplier", reqVO);
        Result<Long> result = supplierRemoteClient.createSupplier(reqVO);
        return handleResult(result, "srm-supplier");
    }
}
```

```java
/**
 * 飞书 API 客户端封装
 */
public abstract class BaseFeishuClient {
    
    protected final Logger log = LoggerFactory.getLogger(getClass());
    
    @Autowired
    private RestTemplate restTemplate;
    
    @Value("${feishu.appId}")
    private String appId;
    
    @Value("${feishu.appSecret}")
    private String appSecret;
    
    /**
     * 获取 tenant_access_token
     */
    protected String getTenantAccessToken() {
        String token = FeishuTokenCache.getTenantAccessToken();
        if (token != null) {
            return token;
        }
        
        FeishuTokenReqVO reqVO = new FeishuTokenReqVO();
        reqVO.setAppId(appId);
        reqVO.setAppSecret(appSecret);
        
        HttpHeaders headers = new HttpHeaders();
        headers.setContentType(MediaType.APPLICATION_JSON);
        HttpEntity<FeishuTokenReqVO> request = new HttpEntity<>(reqVO, headers);
        
        ResponseEntity<FeishuTokenRespVO> response = 
            restTemplate.postForEntity(FeishuApiConstants.GET_TENANT_TOKEN, request, FeishuTokenRespVO.class);
        
        FeishuTokenRespVO respVO = response.getBody();
        if (respVO == null || respVO.getCode() != 0) {
            log.error("获取飞书 access_token 失败: {}", respVO);
            throw new ServiceException("调用飞书接口失败");
        }
        
        // 缓存 token
        FeishuTokenCache.setTenantAccessToken(
            respVO.getTenantAccessToken(), 
            respVO.getExpire() - 300 // 提前5分钟过期
        );
        
        return respVO.getTenantAccessToken();
    }
    
    /**
     * 发送 POST 请求到飞书
     */
    protected <R> R post(String url, Object request, Class<R> responseType) {
        HttpHeaders headers = new HttpHeaders();
        headers.setContentType(MediaType.APPLICATION_JSON);
        headers.setBearerAuth(getTenantAccessToken());
        
        HttpEntity<Object> httpEntity = new HttpEntity<>(request, headers);
        
        log.info("调用飞书 API: url={}, request={}", url, request);
        ResponseEntity<R> response = restTemplate.postForEntity(url, httpEntity, responseType);
        
        log.info("飞书 API 返回: {}", response.getBody());
        return response.getBody();
    }
    
    /**
     * 发送 GET 请求到飞书
     */
    protected <R> R get(String url, Class<R> responseType) {
        HttpHeaders headers = new HttpHeaders();
        headers.setBearerAuth(getTenantAccessToken());
        
        HttpEntity<Void> request = new HttpEntity<>(headers);
        ResponseEntity<R> response = restTemplate.exchange(
            url, HttpMethod.GET, request, responseType);
        
        return response.getBody();
    }
}

/**
 * 飞书审批客户端
 */
@Service
public class FeishuApprovalClient extends BaseFeishuClient {
    
    /**
     * 发起审批实例
     */
    public FeishuCreateApprovalRespVO createApproval(FeishuApprovalCreateReqVO reqVO) {
        String url = FeishuApiConstants.CREATE_APPROVAL_INSTANCE;
        return post(url, reqVO, FeishuCreateApprovalRespVO.class);
    }
    
    /**
     * 查询审批实例详情
     */
    public FeishuApprovalDetailRespVO getApprovalDetail(String instanceId) {

        return get(url, FeishuApprovalDetailRespVO.class);
    }
```

```java
/**
 * 基础 Repository 接口
 */
public interface BaseRepository<T, ID> {
    
    int insert(T entity);
    
    int insertBatch(List<T> entityList);
    
    int updateById(T entity);
    
    int deleteById(ID id);
    
    T selectById(ID id);
    
    List<T> selectByIds(Collection<ID> ids);
    
    List<T> selectList();
}

/**
 * MyBatis Plus 基础实现
 */
public abstract class BaseRepositoryImpl<T, ID, M extends BaseMapper<T>> 
    implements BaseRepository<T, ID> {
    
    @Autowired
    protected M mapper;
    
    @Override
    public int insert(T entity) {
        return mapper.insert(entity);
    }
    
    @Override
    public int insertBatch(List<T> entityList) {
        return entityList.stream()
            .mapToInt(mapper::insert)
            .sum();
    }
    
    @Override
    public int updateById(T entity) {
        return mapper.updateById(entity);
    }
    
    @Override
    public int deleteById(ID id) {
        return mapper.deleteById(id);
    }
    
    @Override
    public T selectById(ID id) {
        return mapper.selectById(id);
    }
    
    @Override
    public List<T> selectByIds(Collection<ID> ids) {
        return mapper.selectBatchIds(ids);
    }
    
    @Override
    public List<T> selectList() {
        return mapper.selectList(null);
    }
}
```

### 5.2 缓存操作封装

```java
/**
 * 缓存服务接口
 */
public interface CacheService {
    
    <T> void set(String key, T value);
    
    <T> void set(String key, T value, long timeout, TimeUnit timeUnit);
    
    <T> T get(String key);
    
    void delete(String key);
    
    boolean exists(String key);
    
    <T> T getWithLoader(String key, Class<T> type, Supplier<T> loader, long timeout, TimeUnit timeUnit);
}

/**
 * Redis 缓存实现
 */
@Service
public class RedisCacheService implements CacheService {
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    private final ObjectMapper objectMapper = new ObjectMapper();
    
    @Override
    public <T> void set(String key, T value) {
        redisTemplate.opsForValue().set(key, value);
    }
    
    @Override
    public <T> void set(String key, T value, long timeout, TimeUnit timeUnit) {
        redisTemplate.opsForValue().set(key, value, timeout, timeUnit);
    }
    
    @Override
    @SuppressWarnings("unchecked")
    public <T> T get(String key) {
        Object value = redisTemplate.opsForValue().get(key);
        if (value == null) {
            return null;
        }
        return (T) value;
    }
    
    @Override
    public void delete(String key) {
        redisTemplate.delete(key);
    }
    
    @Override
    public boolean exists(String key) {
        return Boolean.TRUE.equals(redisTemplate.hasKey(key));
    }
    
    @Override
    public <T> T getWithLoader(String key, Class<T> type, Supplier<T> loader, long timeout, TimeUnit timeUnit) {
        T value = get(key);
        if (value == null) {
            synchronized (key.intern()) {
                value = get(key);
                if (value == null) {
                    value = loader.get();
                    if (value != null) {
                        set(key, value, timeout, timeUnit);
                    }
                }
            }
        }
        return value;
    }
}

/**
 * 缓存 Key 常量定义
 */
public class CacheKeyConstants {
    
    public static final String SUPPLIER_DETAIL = "supplier:detail:%d";
    
    public static final String PRICING_DETAIL = "pricing:detail:%d";
    
    public static final String CONTRACT_TEMPLATE = "contract:template:%s:%s";
    
    public static final String USER_DETAIL = "user:detail:%d";
}

/**
 * 使用示例
 */
@Service
public class SupplierServiceImpl {
    
    @Autowired
    private CacheService cacheService;
    
    public SupplierVO getSupplier(Long id) {
        String cacheKey = String.format(CacheKeyConstants.SUPPLIER_DETAIL, id);
        
        SupplierVO supplierVO = cacheService.getWithLoader(
            cacheKey, 
            SupplierVO.class, 
            () -> {
                SupplierDO supplierDO = supplierMapper.selectById(id);
                if (supplierDO == null) {
                    return null;
                }
                return supplierConverter.toVO(supplierDO);
            },
            1, TimeUnit.HOURS
        );
        
        if (supplierVO == null) {
            throw exception(SUPPLIER_NOT_EXISTS);
        }
        
        return supplierVO;
    }
}
```

```typescript
// src/libs/log/index.ts
import type { Logger, CreateLoggerOptions, LogLevel } from './types';

const LOG_LEVELS: Record<LogLevel, number> = {
  debug: 0,
  info: 1,
  warn: 2,
  error: 3,
};

export function createLogger(options: CreateLoggerOptions): Logger {
  const { name, level = 'info', enableConsole = true, enableRemote = false, remoteUrl } = options;

  const shouldLog = (logLevel: LogLevel): boolean => {
    return LOG_LEVELS[logLevel] >= LOG_LEVELS[level];
  };

  const formatMessage = (logLevel: LogLevel, message: string): string => {
    const timestamp = new Date().toISOString();
    return `[${timestamp}] [${logLevel.toUpperCase()}] [${name}] ${message}`;
  };

  const sendToRemote = async (logLevel: LogLevel, message: string, args: any[]) => {
    if (!enableRemote || !remoteUrl) return;
    try {
      await fetch(remoteUrl, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ name, level: logLevel, message, args, timestamp: Date.now() }),
      });
    } catch (e) {
      console.error('Failed to send log to remote:', e);
    }
  };

  const log = (logLevel: LogLevel, message: string, ...args: any[]) => {
    if (!shouldLog(logLevel)) return;

    const formattedMessage = formatMessage(logLevel, message);

    if (enableConsole) {
      const consoleMethod = logLevel === 'error' ? console.error : 
                            logLevel === 'warn' ? console.warn : 
                            logLevel === 'debug' ? console.debug : console.log;
      consoleMethod(formattedMessage, ...args);
    }

    sendToRemote(logLevel, message, args);
  };

  return {
    debug: (message: string, ...args: any[]) => log('debug', message, ...args),
    info: (message: string, ...args: any[]) => log('info', message, ...args),
    warn: (message: string, ...args: any[]) => log('warn', message, ...args),
    error: (message: string, ...args: any[]) => log('error', message, ...args),
  };
}

// 全局日志实例
export const logger = createLogger({ name: 'app' });
```

```typescript
// src/libs/mock/MockDB.ts
export interface QueryOptions {
  page?: number;
  pageSize?: number;
  sortField?: string;
  sortOrder?: 'asc' | 'desc';
  [key: string]: any;
}

export interface MockDBOptions<T> {
  name: string;
  primaryKey?: keyof T;
}

export class MockDB<T extends Record<string, any>> {
  private data: T[] = [];
  private name: string;
  private primaryKey: keyof T;

  constructor(private options: MockDBOptions<T>) {
    this.name = options.name;
    this.primaryKey = options.primaryKey || ('id' as keyof T);
  }

  setData(data: T[]): void {
    this.data = [...data];
  }

  query(filters?: Partial<T> & QueryOptions): { list: T[]; total: number } {
    let result = [...this.data];

    if (filters) {
      Object.entries(filters).forEach(([key, value]) => {
        if (key === 'page' || key === 'pageSize' || key === 'sortField' || key === 'sortOrder') return;
        if (value !== undefined && value !== null) {
          if (typeof value === 'string' && value.includes('%')) {
            result = result.filter(item => String(item[key]).includes(value.replace('%', '')));
          } else {
            result = result.filter(item => item[key] === value);
          }
        }
      });

      if (filters.sortField) {
        result.sort((a, b) => {
          const valA = a[filters.sortField!];
          const valB = b[filters.sortField!];
          const order = filters.sortOrder === 'asc' ? 1 : -1;
          return valA > valB ? order : -order;
        });
      }
    }

    const total = result.length;
    const page = filters?.page || 1;
    const pageSize = filters?.pageSize || 20;
    const list = result.slice((page - 1) * pageSize, page * pageSize);

    return { list, total };
  }

  findById(id: any): T | undefined {
    return this.data.find(item => item[this.primaryKey] === id);
  }

  insert(item: Omit<T, 'id'> & { id?: any }): T {
    const newItem = { ...item } as T;
    if (!newItem[this.primaryKey]) {
      newItem[this.primaryKey] = Date.now() as any;
    }
    this.data.push(newItem);
    return newItem;
  }

  update(id: any, updates: Partial<T>): T | undefined {
    const index = this.data.findIndex(item => item[this.primaryKey] === id);
    if (index === -1) return undefined;
    this.data[index] = { ...this.data[index], ...updates };
    return this.data[index];
  }

  delete(id: any): boolean {
    const index = this.data.findIndex(item => item[this.primaryKey] === id);
    if (index === -1) return false;
    this.data.splice(index, 1);
    return true;
  }
}
```

```typescript
    if (code === 200) {

    }

    if (code === 401) {
      ElMessage.error('登录已过期，请重新登录');
      window.location.href = '/login';
      return Promise.reject(new Error('登录已过期'));

    return Promise.reject(new Error(message));

    const config = error.config as RequestOptions;

    if (error.response?.status === 401) {
      ElMessage.error('登录已过期，请重新登录');
      window.location.href = '/login';
    } else if (error.response?.status === 403) {
      ElMessage.error('没有权限访问该资源');
    } else if (error.response?.status === 404) {
      ElMessage.error('请求的资源不存在');
    } else if (error.response?.status >= 500) {
      ElMessage.error('服务器错误，请稍后重试');
    } else {
      const message = error.response?.data?.message || error.message || '网络错误';
      if (config?.showError !== false) {
        ElMessage.error(message);
      }
    }
```

```typescript
// src/libs/request/Request.ts
import axios, { AxiosInstance, AxiosRequestConfig } from 'axios';
import type { RequestConfig, RequestOptions, Result } from './types';
import { requestInterceptors, responseInterceptors } from './interceptors';
import { getEnvConfig } from '@/libs/env-config';

export class Request {
  private instance: AxiosInstance;
  private defaultConfig: RequestConfig;

  constructor(config?: Partial<RequestConfig>) {
    const envConfig = getEnvConfig();

    this.defaultConfig = {
      baseURL: envConfig.baseUrl,
      timeout: 30000,
      ...config,
    };

    this.instance = axios.create(this.defaultConfig);
    this.setupInterceptors();
  }

  private setupInterceptors(): void {
    this.instance.interceptors.request.use(
      (config) => requestInterceptors.onRequest(config as RequestOptions),
      requestInterceptors.onRequestError
    );

    this.instance.interceptors.response.use(
      responseInterceptors.onResponse,
      responseInterceptors.onResponseError
    );
  }

  async request<T = any>(config: RequestOptions): Promise<T> {
    const response = await this.instance.request<Result<T>>(config);
    return response.data.data;
  }

  async get<T = any>(url: string, params?: any, config?: RequestOptions): Promise<T> {
    return this.request<T>({ ...config, method: 'GET', url, params });
  }

  async post<T = any>(url: string, data?: any, config?: RequestOptions): Promise<T> {
    return this.request<T>({ ...config, method: 'POST', url, data });
  }

  async put<T = any>(url: string, data?: any, config?: RequestOptions): Promise<T> {
    return this.request<T>({ ...config, method: 'PUT', url, data });
  }

  async delete<T = any>(url: string, params?: any, config?: RequestOptions): Promise<T> {
    return this.request<T>({ ...config, method: 'DELETE', url, params });
  }

  async patch<T = any>(url: string, data?: any, config?: RequestOptions): Promise<T> {
    return this.request<T>({ ...config, method: 'PATCH', url, data });
  }
}

export const request = new Request();
```

```typescript
// src/libs/pdf/generatePDF.ts
import html2canvas from 'html2canvas';
import jsPDF from 'jspdf';

export interface GeneratePDFOptions {
  element: HTMLElement;
  filename?: string;
  orientation?: 'portrait' | 'landscape';
  unit?: 'mm' | 'pt' | 'px';
  format?: 'a3' | 'a4' | 'a5' | 'letter' | 'legal';
  scale?: number;
}

export async function generatePDFFromHTML(options: GeneratePDFOptions): Promise<void> {
  const {
    element,
    filename = 'document.pdf',
    orientation = 'portrait',
    unit = 'mm',
    format = 'a4',
    scale = 2,
  } = options;

  const canvas = await html2canvas(element, {
    scale,
    useCORS: true,
    allowTaint: true,
    backgroundColor: '#ffffff',
  });

  const imgData = canvas.toDataURL('image/png');
  const imgWidth = canvas.width;
  const imgHeight = canvas.height;

  const pdf = new jsPDF(orientation, unit, format);
  const pdfWidth = pdf.internal.pageSize.getWidth();
  const pdfHeight = (imgHeight * pdfWidth) / imgWidth;

  let heightLeft = imgHeight;
  let position = 0;

  pdf.addImage(imgData, 'PNG', 0, position, pdfWidth, pdfHeight);
  heightLeft -= imgHeight;

  while (heightLeft >= 0) {
    position = heightLeft - imgHeight;
    pdf.addPage();
    pdf.addImage(imgData, 'PNG', 0, position, pdfWidth, pdfHeight);
    heightLeft -= imgHeight;
  }

  pdf.save(filename);
}
```

```typescript
  filename?: string;
  method?: 'GET' | 'POST';
  data?: any;
  headers?: Record<string, string>;
}

export async function downloadFile(options: DownloadOptions): Promise<void> {
  const { url, filename, method = 'GET', data, headers = {} } = options;

  if (method === 'GET') {
    const link = document.createElement('a');
    link.href = url;
    link.download = filename || '';
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    return;
  }

  const response = await fetch(url, {
    method,
    headers: { ...headers, 'Content-Type': 'application/json' },
    body: JSON.stringify(data),
  });

  const blob = await response.blob();
  const downloadUrl = window.URL.createObjectURL(blob);
  const link = document.createElement('a');
  link.href = downloadUrl;
  link.download = filename || '';
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
  window.URL.revokeObjectURL(downloadUrl);
}
```

```typescript
// src/libs/db/Storage.ts
export interface StorageOptions {
  prefix?: string;
  expires?: number;
}

export class Storage {
  private prefix: string;

  constructor(options: StorageOptions = {}) {
    this.prefix = options.prefix || 'srm_';
  }

  private getKey(key: string): string {
    return `${this.prefix}${key}`;
  }

  set<T>(key: string, value: T, expires?: number): void {
    const data = {
      value,
      expires: expires ? Date.now() + expires : null,
    };
    localStorage.setItem(this.getKey(key), JSON.stringify(data));
  }

  get<T>(key: string): T | null {
    const raw = localStorage.getItem(this.getKey(key));
    if (!raw) return null;

    try {
      const data = JSON.parse(raw);
      if (data.expires && Date.now() > data.expires) {
        this.remove(key);
        return null;
      }
      return data.value as T;
    } catch {
      return null;
    }
  }

  remove(key: string): void {
    localStorage.removeItem(this.getKey(key));
  }

  clear(): void {
    const keys = Object.keys(localStorage);
    keys.forEach(key => {
      if (key.startsWith(this.prefix)) {
        localStorage.removeItem(key);
      }
    });
  }

  has(key: string): boolean {
    return this.get(key) !== null;
  }
}

export const storage = new Storage({ prefix: 'srm_' });
```

```typescript
// src/libs/db/IndexedDB.ts
export interface IndexedDBOptions {
  name: string;
  version?: number;
  stores?: DBStore[];
}

export interface DBStore {
  name: string;
  keyPath: string;
  indexes?: { name: string; keyPath: string; unique?: boolean }[];
}

export class IndexedDB {
  private db: IDBDatabase | null = null;
  private name: string;
  private version: number;

  constructor(options: IndexedDBOptions) {
    this.name = options.name;
    this.version = options.version || 1;
  }

  async open(): Promise<IDBDatabase> {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(this.name, this.version);

      request.onerror = () => reject(request.error);
      request.onsuccess = () => {
        this.db = request.result;
        resolve(this.db);
      };

      request.onupgradeneeded = (event) => {
        const db = (event.target as IDBOpenDBRequest).result;
        // 在这里创建对象存储
      };
    });
  }

  async add<T>(storeName: string, data: T): Promise<any> {
    if (!this.db) await this.open();
    return new Promise((resolve, reject) => {
      const transaction = this.db!.transaction(storeName, 'readwrite');
      const store = transaction.objectStore(storeName);
      const request = store.add(data);
      request.onsuccess = () => resolve(request.result);
      request.onerror = () => reject(request.error);
    });
  }

  async put<T>(storeName: string, data: T): Promise<any> {
    if (!this.db) await this.open();
    return new Promise((resolve, reject) => {
      const transaction = this.db!.transaction(storeName, 'readwrite');
      const store = transaction.objectStore(storeName);
      const request = store.put(data);
      request.onsuccess = () => resolve(request.result);
      request.onerror = () => reject(request.error);
    });
  }

  async get<T>(storeName: string, key: any): Promise<T | undefined> {
    if (!this.db) await this.open();
    return new Promise((resolve, reject) => {
      const transaction = this.db!.transaction(storeName, 'readonly');
      const store = transaction.objectStore(storeName);
      const request = store.get(key);
      request.onsuccess = () => resolve(request.result);
      request.onerror = () => reject(request.error);
    });
  }

  async delete(storeName: string, key: any): Promise<void> {
    if (!this.db) await this.open();
    return new Promise((resolve, reject) => {
      const transaction = this.db!.transaction(storeName, 'readwrite');
      const store = transaction.objectStore(storeName);
      const request = store.delete(key);
      request.onsuccess = () => resolve();
      request.onerror = () => reject(request.error);
    });
  }

  async getAll<T>(storeName: string): Promise<T[]> {
    if (!this.db) await this.open();
    return new Promise((resolve, reject) => {
      const transaction = this.db!.transaction(storeName, 'readonly');
      const store = transaction.objectStore(storeName);
      const request = store.getAll();
      request.onsuccess = () => resolve(request.result);
      request.onerror = () => reject(request.error);
    });
  }

  async clear(storeName: string): Promise<void> {
    if (!this.db) await this.open();
    return new Promise((resolve, reject) => {
      const transaction = this.db!.transaction(storeName, 'readwrite');
      const store = transaction.objectStore(storeName);
      const request = store.clear();
      request.onsuccess = () => resolve();
      request.onerror = () => reject(request.error);
    });
  }
}
```

## 8. 总结

| 封装层级 | 职责 | 对应文件 |
|---------|------|---------|
| **数据封装** | DO/VO/ReqVO/Convert | framework-common 库 + 各模块 |
| **网络请求** | 统一响应、Feign、第三方API | framework-web + framework-integration |
| **数据访问** | Repository 基类、缓存封装 | framework-data 库 |
| **环境配置** | Profile 配置、环境检测 | 配置文件 + EnvironmentHelper |
| **模块引用** | 明确的依赖关系 | 各模块的 POM 文件 |

这样的封装设计确保了：
- ✅ 代码分层清晰，职责明确
- ✅ 各模块高内聚低耦合
- ✅ 基础设施层可复用
- ✅ 跨环境部署灵活配置
- ✅ 第三方集成统一管理
