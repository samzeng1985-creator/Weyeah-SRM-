```java
package com.srm.supplier.controller;

@RestController
@RequestMapping("/api/supplier/suppliers")
@RequiredArgsConstructor
@Slf4j
@Tag(name = "供应商管理", description = "供应商相关接口")
public class SupplierController {

    private final SupplierService supplierService;

    /**
     * 分页查询供应商列表
     */
    @GetMapping
    @Operation(summary = "查询供应商列表", description = "分页查询供应商信息")
    @PreAuthorize("@ss.hasPermission('supplier:list')")
    public Result<PageResult<SupplierRespVO>> getSupplierPage(
            @Valid SupplierPageReqVO reqVO) {
        log.info("查询供应商列表 | page={}, pageSize={}, keyword={}", 
                reqVO.getPage(), reqVO.getPageSize(), reqVO.getKeyword());
        return Result.success(supplierService.getSupplierPage(reqVO));
    }

    /**
     * 获取供应商详情
     */
    @GetMapping("/{id}")
    @Operation(summary = "获取供应商详情")
    @PreAuthorize("@ss.hasPermission('supplier:query')")
    public Result<SupplierRespVO> getSupplier(@PathVariable Long id) {
        log.info("获取供应商详情 | id={}", id);
        return Result.success(supplierService.getSupplier(id));
    }

    /**
     * 创建供应商
     */
    @PostMapping
    @Operation(summary = "创建供应商")
    @PreAuthorize("@ss.hasPermission('supplier:create')")
    public Result<Long> createSupplier(@Valid @RequestBody SupplierCreateReqVO reqVO) {
        log.info("创建供应商 | supplierName={}", reqVO.getSupplierName());
        Long id = supplierService.createSupplier(reqVO);
        return Result.success(id);
    }

    /**
     * 更新供应商
     */
    @PutMapping("/{id}")
    @Operation(summary = "更新供应商")
    @PreAuthorize("@ss.hasPermission('supplier:update')")
    public Result<Boolean> updateSupplier(
            @PathVariable Long id,
            @Valid @RequestBody SupplierUpdateReqVO reqVO) {
        log.info("更新供应商 | id={}", id);
        supplierService.updateSupplier(id, reqVO);
        return Result.success(true);
    }

    /**
     * 删除供应商
     */
    @DeleteMapping("/{id}")
    @Operation(summary = "删除供应商")
    @PreAuthorize("@ss.hasPermission('supplier:delete')")
    public Result<Boolean> deleteSupplier(@PathVariable Long id) {
        log.info("删除供应商 | id={}", id);
        supplierService.deleteSupplier(id);
        return Result.success(true);
    }
}
```

#### 1.2.2 Service层

```java
package com.srm.supplier.service;

@Service
@RequiredArgsConstructor
@Slf4j
@Transactional(rollbackFor = Exception.class)
public class SupplierServiceImpl implements SupplierService {

    private final SupplierRepository supplierRepository;
    private final SupplierQualificationService qualificationService;
    private final IdGenerator idGenerator;

    @Override
    public PageResult<SupplierRespVO> getSupplierPage(SupplierPageReqVO reqVO) {
        // 1. 查询数据
        Page<SupplierDO> page = supplierRepository.selectPage(reqVO);
        
        // 2. 转换VO
        List<SupplierRespVO> list = page.getRecords().stream()
                .map(SupplierConvert.INSTANCE::toRespVO)
                .collect(Collectors.toList());
        
        // 3. 返回结果
        return new PageResult<>(list, page.getTotal());
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public Long createSupplier(SupplierCreateReqVO reqVO) {
        // 1. 校验供应商编码唯一性
        validateSupplierCodeUnique(reqVO.getSupplierCode());
        
        // 2. 校验供应商类型
        validateSupplierType(reqVO.getSupplierType());
        
        // 3. 生成ID
        Long id = idGenerator.nextId();
        
        // 4. 转换DO
        SupplierDO supplier = SupplierConvert.INSTANCE.toDO(reqVO);
        supplier.setId(id);
        supplier.setStatus(SupplierStatusEnum.PENDING.getStatus());
        
        // 5. 保存数据
        supplierRepository.insert(supplier);
        
        // 6. 记录日志
        log.info("创建供应商成功 | id={}, supplierName={}", id, reqVO.getSupplierName());
        
        return id;
    }

    @Override
    public SupplierRespVO getSupplier(Long id) {
        SupplierDO supplier = supplierRepository.selectById(id);
        if (supplier == null) {
            throw exception(SUPPLIER_NOT_EXISTS);
        }
        return SupplierConvert.INSTANCE.toRespVO(supplier);
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void updateSupplier(Long id, SupplierUpdateReqVO reqVO) {
        // 1. 校验供应商存在
        SupplierDO supplier = validateSupplierExists(id);
        
        // 2. 校验供应商状态（已生效的不能随意修改）
        validateSupplierEditable(supplier);
        
        // 3. 更新数据
        SupplierConvert.INSTANCE.updateFromReqVO(supplier, reqVO);
        supplierRepository.update(supplier);
        
        // 4. 记录日志
        log.info("更新供应商成功 | id={}", id);
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public void deleteSupplier(Long id) {
        // 1. 校验供应商存在
        SupplierDO supplier = validateSupplierExists(id);
        
        // 2. 校验是否可以删除（有未完成合同时不能删除）
        validateSupplierDeletable(id);
        
        // 3. 删除数据
        supplierRepository.deleteById(id);
        
        // 4. 记录日志
        log.info("删除供应商成功 | id={}", id);
    }

    /**
     * 校验供应商编码唯一性
     */
    private void validateSupplierCodeUnique(String supplierCode) {
        SupplierDO existing = supplierRepository.selectBySupplierCode(supplierCode);
        if (existing != null) {
            throw exception(SUPPLIER_CODE_DUPLICATE, supplierCode);
        }
    }

    /**
     * 校验供应商存在
     */
    private SupplierDO validateSupplierExists(Long id) {
        SupplierDO supplier = supplierRepository.selectById(id);
        if (supplier == null) {
            throw exception(SUPPLIER_NOT_EXISTS);
        }
        return supplier;
    }
}
```

**Service层规范：**
- ✅ 使用 `@Transactional` 控制事务
- ✅ 所有异常使用 `exception()` 方法抛出
- ✅ 日志记录关键操作和结果
- ✅ 使用 `Convert` 类转换对象
- ✅ 复杂业务拆分成私有方法
- ✅ 校验逻辑单独封装

#### 1.2.3 Repository层

```java
package com.srm.supplier.dal.mysql;

@Mapper
@Repository
public interface SupplierRepository {

    /**
     * 分页查询
     */
    default Page<SupplierDO> selectPage(SupplierPageReqVO reqVO) {
        return new LambdaQueryWrapperX<SupplierDO>()
                .likeIfPresent(SupplierDO::getSupplierName, reqVO.getKeyword())
                .eqIfPresent(SupplierDO::getSupplierType, reqVO.getSupplierType())
                .eqIfPresent(SupplierDO::getStatus, reqVO.getStatus())
                .betweenIfPresent(SupplierDO::getCreatedTime, 
                        reqVO.getStartTime(), reqVO.getEndTime())
                .orderByDesc(SupplierDO::getCreatedTime)
                .page(reqVO);
    }

    /**
     * 根据ID查询
     */
    SupplierDO selectById(@Param("id") Long id);

    /**
     * 根据编码查询
     */
    SupplierDO selectBySupplierCode(@Param("supplierCode") String supplierCode);

    /**
     * 批量插入
     */
    int insertBatch(@Param("list") List<SupplierDO> list);

    /**
     * 更新
     */
    int update(SupplierDO supplier);

    /**
     * 删除
     */
    int deleteById(@Param("id") Long id);
}
```

```java
package com.srm.framework.common.po;

@Data
@NoArgsConstructor
@AllArgsConstructor
public class Result<T> implements Serializable {
    
    private static final long serialVersionUID = 1L;
    
    /** 成功码 */
    public static final int SUCCESS_CODE = 200;
    /** 业务错误码 */
    public static final int ERROR_CODE = 500;
    
    /** 状态码 */
    private int code;
    /** 提示信息 */
    private String message;
    /** 数据 */
    private T data;
    
    /** 成功结果 */
    public static <T> Result<T> success(T data) {
        return new Result<>(SUCCESS_CODE, "操作成功", data);
    }
    
    /** 成功结果 */
    public static <T> Result<T> success(String message, T data) {
        return new Result<>(SUCCESS_CODE, message, data);
    }
    
    /** 错误结果 */
    public static <T> Result<T> error(String message) {
        return new Result<>(ERROR_CODE, message, null);
    }
    
    /** 错误结果 */
    public static <T> Result<T> error(int code, String message) {
        return new Result<>(code, message, null);
    }
}
```

```typescript
import { ElMessage, ElMessageBox } from 'element-plus'
import router from '@/router'

const service = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 30000
})

// 请求拦截器
service.interceptors.request.use(
  (config) => {
    // 添加Token
    const token = localStorage.getItem('token')
    if (token) {
      config.headers['Authorization'] = `Bearer ${token}`
    }
    
    // 添加时间戳，防止缓存
    config.params = {
      ...config.params,
      _t: Date.now()
    }
    
    return config
  },
  (error) => {
    return Promise.reject(error)
  }
)

// 响应拦截器
service.interceptors.response.use(
  (response: AxiosResponse) => {
    const res = response.data
    
    // 成功
    if (res.code === 200) {
      return res.data
    }
    
    // 未登录
    if (res.code === 401) {
      ElMessageBox.confirm('登录已过期，请重新登录', '提示', {
        confirmButtonText: '重新登录',
        cancelButtonText: '取消',
        type: 'warning'
      }).then(() => {
        localStorage.removeItem('token')
        router.push('/login')
      })
      return Promise.reject(new Error('登录已过期'))
    }
    
    // 无权限
    if (res.code === 403) {
      ElMessage.error('没有权限访问该资源')
      return Promise.reject(new Error('没有权限'))
    }
    
    // 其他错误
    ElMessage.error(res.message || '请求失败')
    return Promise.reject(new Error(res.message))
  },
  (error) => {
    // 网络错误
    if (!error.response) {
      ElMessage.error('网络连接失败，请检查网络')
      return Promise.reject(error)
    }
    
    ElMessage.error(error.response.data?.message || '请求失败')
    return Promise.reject(error)
  }
)

export default service
```

```typescript
export const useSupplierStore = defineStore('supplier', {
  state: (): SupplierState => ({
    list: [],
    total: 0,
    loading: false,
    currentSupplier: null
  }),
  
  actions: {
    async fetchList(params: SupplierPageReqVO) {
      this.loading = true
      try {
        const result = await getSupplierPage(params)
        this.list = result.records
        this.total = result.total
      } finally {
        this.loading = false
      }
    },
    
    async fetchCurrent(id: number) {
      this.loading = true
      try {
        this.currentSupplier = await getSupplier(id)
      } finally {
        this.loading = false
      }
    },
    
    clearCurrent() {
      this.currentSupplier = null
    }
  }
})
```
